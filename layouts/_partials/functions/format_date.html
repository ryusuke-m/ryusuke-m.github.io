{{/* Format a date according to the page's `date_precision` front matter (day | month | year). Inputs: dict "date" "precision" Output: string. Falls back to the site-wide date format for `day` or unset. Note: a single `return` at the very end keeps this compatible with older Hugo versions. */}} {{- $date := .date -}} {{- $lang := site.Language.Lang -}} {{- $out := time.Format (site.Params.hugoblox.locale.date_format | default ":date_long") $date -}} {{- if eq .precision "month" -}} {{- $out = cond (eq $lang "ja") (time.Format "2006年1月" $date) (time.Format "January 2006" $date) -}} {{- else if eq .precision "year" -}} {{- $out = cond (eq $lang "ja") (time.Format "2006年" $date) (time.Format "2006" $date) -}} {{- end -}} {{- return $out -}}
